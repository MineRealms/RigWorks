cmake_minimum_required(VERSION 3.21)
project(rigworks_ysm VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

set(RWYM_ZSTD_SOURCE_DIR "" CACHE PATH "Optional existing zstd 1.5.7 source tree")
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_CONTRIB OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_COMPRESSION OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_DICTBUILDER OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
if(RWYM_ZSTD_SOURCE_DIR)
    set(rwym_zstd_source "${RWYM_ZSTD_SOURCE_DIR}")
    add_subdirectory("${RWYM_ZSTD_SOURCE_DIR}/build/cmake" "${CMAKE_BINARY_DIR}/zstd" EXCLUDE_FROM_ALL)
else()
    include(FetchContent)
    FetchContent_Declare(zstd
        GIT_REPOSITORY https://github.com/facebook/zstd.git
        GIT_TAG ac66b19e6bd6b83238bf008eecc1298105298532
        SOURCE_SUBDIR build/cmake)
    FetchContent_MakeAvailable(zstd)
    set(rwym_zstd_source "${zstd_SOURCE_DIR}")
endif()

add_library(rigworks_ysm_core STATIC src/ysm_v3.cpp src/ysm_payload.cpp src/rwym.cpp)
target_include_directories(rigworks_ysm_core PUBLIC src)
target_link_libraries(rigworks_ysm_core PRIVATE libzstd_static)
set_target_properties(rigworks_ysm_core PROPERTIES CXX_VISIBILITY_PRESET hidden)

add_library(rigworks_ysm SHARED src/rigworks_ysm.cpp)
target_include_directories(rigworks_ysm PUBLIC include)
target_compile_definitions(rigworks_ysm PRIVATE RIGWORKS_YSM_BUILD)
target_link_libraries(rigworks_ysm PRIVATE rigworks_ysm_core)
set_target_properties(rigworks_ysm PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(rigworks_ysm_core PRIVATE /W4 /permissive-)
    target_compile_options(rigworks_ysm PRIVATE /W4 /permissive-)
elseif(APPLE)
    target_link_options(rigworks_ysm PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/exports.macos")
else()
    target_link_options(rigworks_ysm PRIVATE "LINKER:--version-script,${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map")
endif()

set(RWYM_RESOURCE_PREFIX "" CACHE STRING "JNA platform prefix used for staged runtime binaries")
if(NOT RWYM_RESOURCE_PREFIX)
    string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR};${CMAKE_GENERATOR_PLATFORM};${CMAKE_OSX_ARCHITECTURES}" architecture)
    if(architecture MATCHES "aarch64|arm64")
        set(rwym_arch aarch64)
    elseif(architecture MATCHES "x86_64|amd64|x64")
        set(rwym_arch x86-64)
    else()
        message(FATAL_ERROR "Unsupported architecture; set RWYM_RESOURCE_PREFIX explicitly for a new target")
    endif()
    if(WIN32)
        set(RWYM_RESOURCE_PREFIX "win32-${rwym_arch}")
    elseif(APPLE)
        set(RWYM_RESOURCE_PREFIX "darwin-${rwym_arch}")
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(RWYM_RESOURCE_PREFIX "linux-${rwym_arch}")
    else()
        message(FATAL_ERROR "Unsupported system; set RWYM_RESOURCE_PREFIX explicitly for a new target")
    endif()
endif()
message(STATUS "RigWorks YSM runtime resource prefix: ${RWYM_RESOURCE_PREFIX}")
install(TARGETS rigworks_ysm
    RUNTIME DESTINATION "${RWYM_RESOURCE_PREFIX}" COMPONENT RigWorksYsm
    LIBRARY DESTINATION "${RWYM_RESOURCE_PREFIX}" COMPONENT RigWorksYsm)
install(FILES include/rigworks_ysm.h DESTINATION include COMPONENT RigWorksYsm)
install(FILES "${rwym_zstd_source}/LICENSE" DESTINATION licenses
    RENAME zstd-LICENSE COMPONENT RigWorksYsm)

include(CTest)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    set(RWYM_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/test/resources/ysm/Flandre_Scarlet.ysm"
        CACHE FILEPATH "Real legacy V3 fixture used by converter tests")
    if(NOT EXISTS "${RWYM_FIXTURE}")
        message(FATAL_ERROR "RWYM_FIXTURE is missing: ${RWYM_FIXTURE}")
    endif()
    add_executable(ysm_rwym_fixture_test tests/ysm_rwym_fixture_test.cpp)
    target_link_libraries(ysm_rwym_fixture_test PRIVATE rigworks_ysm_core)
    add_executable(ysm_abi_test tests/ysm_abi_test.cpp tests/ysm_abi_c_header_test.c)
    target_link_libraries(ysm_abi_test PRIVATE rigworks_ysm rigworks_ysm_core Threads::Threads)
    add_test(NAME rwym.fixture COMMAND ysm_rwym_fixture_test "${RWYM_FIXTURE}")
    add_test(NAME rwym.c_abi COMMAND ysm_abi_test "${RWYM_FIXTURE}")
endif()
